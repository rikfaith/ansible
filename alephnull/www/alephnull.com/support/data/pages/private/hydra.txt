===== Kernel Recompile =====

  apt-get install linux-image-2.6.38-8-server
  reboot
  apt-get install linux-headers-`uname -r`

  apt-get install build-essential autoconf automake libtool flex bison gdb
  apt-get build-dep linux
  apt-get build-dep --no-install-recommends linux-image-$(uname -r)
  apt-get install libncurses5 libncurses5-dev
  cd
  apt-get source linux-image-$(uname -r)
  cd linux-2.6.38
  patch -p1 < ../mvsas.diff
  cp /boot/config-$(uname -r) .config
  make oldconfig
  cd drivers/scsi/mvsas
  make -C /usr/src/linux-headers-`uname -r` M=`pwd`
  mv /lib/modules/$(uname -r)/kernel/drivers/scsi/mvsas/mvsas.ko /lib/modules/$(uname -r)/kernel/drivers/scsi/mvsas/mvsas_old.ko
  cp mvsas.ko /lib/modules/$(uname -r)/kernel/drivers/scsi/mvsas/mvsas.ko
  cd ../libsas
  make -C /usr/src/linux-headers-`uname -r` M=`pwd`
  mv /lib/modules/$(uname -r)/kernel/drivers/scsi/libsas/libsas.ko /lib/modules/$(uname -r)/kernel/drivers/scsi/libsas/libsas_old.ko
  cp libsas.ko /lib/modules/$(uname -r)/kernel/drivers/scsi/libsas/libsas.ko
  update-initramfs -u -k $(uname -r)
  update-grub

See [[https://bugzilla.redhat.com/attachment.cgi?id=505977 | patch ]] (not used with 8 kernel).

=====setup=====
  for i in b c d e f g h i j k; do parted /dev/sd$i unit kib mkpart primary ext2 28 2930266567 ; done
  for i in b c d e f g h i j k; do parted /dev/sd$i set 1 raid on; done
  for i in b c d e f g h i j k; do parted /dev/sd$i unit kib print; done
  mdadm -C /dev/md0 /dev/sd[bcdef]1 -c 128 -n 5 -l 5
  mdadm -C /dev/md1 /dev/sd[ghijk]1 -c 128 -n 5 -l 5
  cryptsetup -c aes-cbc-essiv:sha256 -s 128 -h sha256 --align-payload=1024 luksFormat /dev/md0
  cryptsetup -c aes-cbc-essiv:sha256 -s 128 -h sha256 --align-payload=1024 luksFormat /dev/md1
  cryptsetup luksOpen /dev/md0 r0
  cryptsetup luksOpen /dev/md1 r1
  pvcreate --metadatasize 506k /dev/mapper/r0
  pvs /dev/mapper/r0 -o+pe_start
  vgcreate -s 1g v0 /dev/mapper/r0
  pvcreate --metadatasize 506k /dev/mapper/r1
  pvs /dev/mapper/r1 -o+pe_start
  vgcreate -s 1g v1 /dev/mapper/r1
