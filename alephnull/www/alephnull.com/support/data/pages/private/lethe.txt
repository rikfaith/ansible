===== Driver Setup =====

  apt-get install build-essential autoconf automake libtool flex bison gdb
  apt-get build-dep linux
  apt-get build-dep --no-install-recommends linux-image-$(uname -r)
  apt-get install libncurses5 libncurses5-dev
  cd
  apt-get source linux-image-$(uname -r)
  cd linux-2.6.31
  ? cp /boot/config-2.6.31-20-server .config
  ? make oldconfig
  ? make menuconfig
  cd drivers/scsi/mvsas
  patch -p1 < ~/mvsas.diff2
  make -C /usr/src/linux-headers-`uname -r` M=`pwd`
  mv /lib/modules/$(uname -r)/kernel/drivers/scsi/mvsas/mvsas.ko /lib/modules/$(uname -r)/kernel/drivers/scsi/mvsas/mvsas_old.ko
  cp mvsas.ko /lib/modules/$(uname -r)/kernel/drivers/scsi/mvsas/mvsas.ko
  update-initramfs -u -k $(uname -r)
