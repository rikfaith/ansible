Use
  fdisk -H 224 -S 56
to make one partition per drive.

Create new RAID5:
  mdadm -C /dev/md_d0 -c 128 -n 5 -l 5 /dev/sd[bcdef]1

Setup crypto:
  cryptsetup -c aes-cbc-essiv:sha256 -s 128 -h sha256 --align-payload=1024 luksFormat /dev/md_d0

Setup LVM:
  cryptsetup luksOpen /dev/md_d0 r0
  pvcreate --metadatasize 506k /dev/mapper/r0 
  pvs /dev/mapper/r0 -o+pe_start
  
  PV         VG   Fmt  Attr PSize PFree 1st PE 
  /dev/dm-0       lvm2 a-   7.28t 7.28t 512.00k
  
  vgcreate -s 1g v0 /dev/mapper/r0
  
Setup partitions:
  lvcreate -L 7t -n tv v0
  mke2fs -t ext4 -i1048576 -m0 -E stride=32,stripe-width=128 /dev/mapper/v0-tv
  mkdir /tv
  mount -o noatime,nodiratime /dev/mapper/v0-tv /tv
